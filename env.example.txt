# Copy this file to .env and fill in what you need.
# .env must never be committed — check that .gitignore contains it.

# ---- required ----------------------------------------------------------

# From console.anthropic.com. The only setting ARC genuinely can't run without.
ANTHROPIC_API_KEY=

# ---- sign-in -----------------------------------------------------------

# Who may sign in. Google proves who somebody is; this decides whether that
# somebody is you. Comma-separated. Leave it empty and ARC refuses to start,
# because coming up with no gate would be worse than not coming up at all.
ARC_ALLOWED_EMAILS=you@gmail.com

# Guests. Listed here they sign in normally (no need to repeat the address
# above) but get a cut-down ARC: their own Google account plus weather, stocks,
# news and search. No Telegram, notes, memory, todos, reminders, phone alerts,
# second screen or computer control. For people you want to show ARC to rather
# than hand it to. See GUEST_TOOLS in run.py for the exact list.
ARC_GUEST_EMAILS=

# google (default) — Google OAuth, the only sane setting for anything
#                    reachable off this machine.
# open            — no sign-in at all. Refused unless ARC_HOST is loopback.
ARC_AUTH_MODE=google

# The owner (the address above, minus any guests) has no session timeout at
# all: no idle clock, no cap, and closing the tab is not a sign-out. Timeouts
# stop a session outliving the person using it, which is a guest's phone, not
# your own desk. 0 puts everyone on the same clocks.
# Which language your ElevenLabs voice actually sounds like. Its model is
# multilingual but its mouth is not, so when ARC is speaking another language
# it uses the free Edge voice for that language instead. Only matters if your
# ElevenLabs voice is not English.
ARC_ELEVEN_LANG=en

ARC_OWNER_SESSION_UNLIMITED=1

# Live sign-ins one address may hold at once. A ninth drops whichever has gone
# longest unused, deleting its Google token with it — worth having because an
# owner session never expires on its own. 0 = no limit.
ARC_MAX_SESSIONS_PER_ACCOUNT=8

# How long a GUEST sign-in lasts. Two clocks, and the session dies when either
# runs out. The 4-hour cap is a ceiling: a larger value is capped, not obeyed.
# 0 turns the cap off and leaves the idle clock as the only one.
ARC_SESSION_MAX_HOURS=4
ARC_SESSION_IDLE_MINUTES=30

# While an alarm is set, its poll counts as use so the tab is still signed
# in when the alarm goes off. 0 = strict idle timeout; alarms then still
# fire and still reach your phone, but won't ring in a tab left open.
ARC_ALARM_KEEPS_SESSION=1


# The origin Google should redirect back to, e.g. https://your-host.ts.net
# Pin this on anything public: unset, it is derived from forwarding headers a
# client can set. Leave blank on the desktop.
ARC_PUBLIC_URL=

# Signs the short-lived sign-in cookie. A long random string — mash the
# keyboard. Without it, a restart mid-sign-in fails and has to be retried.
ARC_SECRET=

# ---- optional: voice ---------------------------------------------------

# Without these, ARC uses the browser's built-in voice. It works, it just
# sounds like a browser.
ELEVENLABS_API_KEY=
ELEVENLABS_VOICE_ID=

# ---- optional: model ---------------------------------------------------

# claude-haiku-4-5 is faster and cheaper if you want a snappier loop.
ARC_MODEL=claude-sonnet-5
# Covers thinking AND the reply together. Don't drop this below a few thousand
# or complex answers come back truncated (or empty, which reads as silence).
ARC_MAX_TOKENS=8000
# low | medium | high — how hard ARC thinks before answering.
ARC_EFFORT=medium

# ---- optional: capabilities --------------------------------------------

# Calendar and Gmail need no env vars — they use credentials.json + a browser
# sign-in (python gauth.py). See the README.

# Gmail sending brakes. The allowlist, if set, means ARC can email nobody else.
# (ARC_MAX_SENDS_PER_HOUR and ARC_EMAIL_ALLOWLIST are gone: Gmail is
#  read-only now, so ARC cannot send mail at all and there is nothing to cap.)

# Telegram (python tg_login.py). Get these two from https://my.telegram.org
# -> API development tools. The login itself is a phone + SMS code.
TG_API_ID=
TG_API_HASH=
ARC_TG_MAX_SENDS_PER_HOUR=10
# Name/username fragments; if set, ARC can only message these chats.
ARC_TG_ALLOWLIST=

# ---- optional: limits --------------------------------------------------

# Per person, and then across the whole deployment. These are the brakes that
# stop one bad afternoon becoming a large bill.
ARC_RATE_PER_MIN=12
ARC_RATE_PER_HOUR=120
ARC_DAILY_CAP=600

# ---- optional: local only ----------------------------------------------

# Ignored when deployed — hosts supply their own PORT.
ARC_PORT=8420

# Which network interface to bind. Default 127.0.0.1 (this machine only).
# Set 0.0.0.0 to let other devices on your Wi-Fi (e.g. your phone at
# http://<this-PC-IP>:8420) reach it — only with ARC_AUTH_MODE left at the
# google default, never with it set to open. Note: voice
# needs a secure origin, so over plain http a phone gets text only; for voice
# on a phone you need HTTPS (see the README "On your phone" section).
ARC_HOST=127.0.0.1
